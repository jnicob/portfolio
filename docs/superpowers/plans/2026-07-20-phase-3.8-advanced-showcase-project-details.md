/**
 * Re-aplica la apariencia tras un remount del root layout (p.ej. cambio de locale):
 * React re-aplica el data-theme="dark" estático del <html> pisando la elección del
 * usuario. Storage manda (applyTheme/applySkin lo actualizan en cada cambio); el
 * fallback es la resolución de la primera carga (para storage no disponible/ inválido).
 */
export function reapplyStoredAppearance(fallback: Appearance): void {
  let storedTheme: string | null = null;
  let storedSkin: string | null = null;
  try {
    storedTheme = localStorage.getItem(STORAGE_KEYS.theme);
    storedSkin = localStorage.getItem(STORAGE_KEYS.skin);
  } catch {
    /* almacenamiento no disponible: re-aplica el fallback de la primera carga */
  }
  applyAppearance({
    theme: parseValid(themeSchema, storedTheme) ?? fallback.theme,
    skin: parseValid(skinSchema, storedSkin) ?? fallback.skin,
  });
}
```

/**
 * Motor de layouts de FilterGallery (spec F3.8 §2): funciones PURAS que derivan
 * posiciones {x,y,width,height} de aspect ratios conocidos — nunca miden el DOM.
 * La posición es solo visual: el orden DOM (= orden de datos = orden de lectura)
 * lo mantiene FilterGallery. Patrón clean-room de los masonry internos analizados
 * en el brainstorm (packing determinista «columna más corta primero»).
 */

export type LayoutBox = { x: number; y: number; width: number; height: number };
export type ComputedLayout = { boxes: readonly LayoutBox[]; totalHeight: number };

/** Ratio inválido (≤0, NaN, Infinity) → 1 (cuadrado), documentado en el README. */
function normalizeRatio(ratio: number): number {
  return Number.isFinite(ratio) && ratio > 0 ? ratio : 1;
}

export function computeMasonryLayout(input: {
  aspectRatios: readonly number[];
  containerWidth: number;
  columns: number;
  gap: number;
  extraHeight?: number;
}): ComputedLayout {
  const { aspectRatios, containerWidth, columns, gap, extraHeight = 0 } = input;
  if (aspectRatios.length === 0 || containerWidth <= 0 || columns <= 0) {
    return { boxes: [], totalHeight: 0 };
  }
  const width = (containerWidth - gap * (columns - 1)) / columns;
  // nextY[i] = próxima y libre de la columna i (ya incluye el gap tras cada caja).
  const nextY = Array.from({ length: columns }, () => 0);
  const boxes = aspectRatios.map((ratio) => {
    const column = nextY.indexOf(Math.min(...nextY));
    const height = width / normalizeRatio(ratio) + extraHeight;
    const box = { x: column * (width + gap), y: nextY[column], width, height };
    nextY[column] = box.y + height + gap;
    return box;
  });
  return { boxes, totalHeight: Math.max(...nextY) - gap };
}
```

```ts
/**
 * Filas justificadas (patrón Flickr/Google Photos): greedy — se acumulan ítems a
 * alto objetivo hasta alcanzar el ancho del contenedor y la fila se escala para
 * llenarlo exacto; la última fila incompleta conserva el alto objetivo (no se estira).
 */
export function computeJustifiedLayout(input: {
  aspectRatios: readonly number[];
  containerWidth: number;
  targetRowHeight: number;
  gap: number;
  extraHeight?: number;
}): ComputedLayout {
  const { aspectRatios, containerWidth, targetRowHeight, gap, extraHeight = 0 } = input;
  if (aspectRatios.length === 0 || containerWidth <= 0) return { boxes: [], totalHeight: 0 };

  const ratios = aspectRatios.map(normalizeRatio);
  const rows: number[][] = [];
  let current: number[] = [];
  let widthAtTarget = 0;
  ratios.forEach((ratio, index) => {
    current.push(index);
    widthAtTarget += targetRowHeight * ratio;
    if (widthAtTarget + gap * (current.length - 1) >= containerWidth) {
      rows.push(current);
      current = [];
      widthAtTarget = 0;
    }
  });
  const lastIsPartial = current.length > 0;
  if (lastIsPartial) rows.push(current);

  const boxes: LayoutBox[] = Array.from({ length: ratios.length }) as LayoutBox[];
  let y = 0;
  rows.forEach((row, rowIndex) => {
    const gaps = gap * (row.length - 1);
    const rowRatioSum = row.reduce((sum, i) => sum + ratios[i], 0);
    const isLast = rowIndex === rows.length - 1 && lastIsPartial;
    // Filas completas: alto que llena el ancho exacto. Última parcial: alto objetivo.
    const mediaHeight = isLast ? targetRowHeight : (containerWidth - gaps) / rowRatioSum;
    let x = 0;
    for (const i of row) {
      const width = mediaHeight * ratios[i];
      boxes[i] = { x, y, width, height: mediaHeight + extraHeight };
      x += width + gap;
    }
    y += mediaHeight + extraHeight + gap;
  });
  return { boxes, totalHeight: y - gap };
}

/**
 * Columnas del masonry por ancho de CONTENEDOR (no de viewport): 2 → 3 → 4 → 5.
 * Umbrales elegidos para tiles ≥ ~230px; máximo 5 (spec F3.8: 4-5 columnas máx).
 */
export const MASONRY_COLUMN_THRESHOLDS = [
  { minWidth: 1200, columns: 5 },
  { minWidth: 880, columns: 4 },
  { minWidth: 560, columns: 3 },
] as const;

export function columnCountForWidth(containerWidth: number): number {
  for (const { minWidth, columns } of MASONRY_COLUMN_THRESHOLDS) {
    if (containerWidth >= minWidth) return columns;
  }
  return 2;
}
```

description: Use when building or reviewing gallery layouts — masonry, justified rows, responsive column counts, or any variable-height media grid

# Masonry y layouts de galería

## Regla nº 1: CSS multi-column NO es masonry de producto

`columns:`/`column-count` llena columna a columna (column-major): el orden de
lectura/Tab deja de coincidir con el visual (rompe a11y), los ítems saltan de
columna al filtrar y no hay control de posiciones. Los masonry de producción
(patrón validado en galerías a gran escala) usan **JS de columnas balanceadas**.

## El patrón

1. **Motor puro**: `(aspectRatios, containerWidth, columns, gap) → {x,y,width,height}[]`.
   Packing determinista «columna más corta primero». Sin medir el DOM: las
   alturas se derivan de `width/height` CONOCIDOS del asset (exigirlos en los datos).
2. **Orden DOM = orden de datos**: la posición es solo visual (`position:absolute`
   + `left/top`); nunca reordenar el DOM ni repartir en wrappers por columna
   (eso rompe el orden de lectura).
3. **Columnas por ancho de CONTENEDOR** (ResizeObserver), no de viewport:
   umbrales explícitos y testeados (aquí: 2 → 3 → 4 → 5, máx 5).
4. **Justified = mismo motor por filas**: alto objetivo, anchos ∝ aspect ratio,
   la fila escala para llenar el ancho exacto; la última fila no se estira.
5. **Animación**: FLIP con `transform` compone por encima de `left/top` sin
   conflicto. `prefers-reduced-motion` → recolocación instantánea.
6. **SSR/hidratación**: no medir en servidor; aplicar posiciones solo tras la
   primera medición client-side (el HTML prerenderizado queda en el layout CSS
   por defecto).

## En este repo

- Motor: `packages/media-kit/src/filter-gallery/layout-engine.ts` (tests al lado).
- Aplicación: prop `layout` de `FilterGallery` (`filter-gallery.tsx`).
- Consumidor: `apps/web/src/components/showcase/gallery-demo.tsx` (selector).
- Decisión y análisis de referencia: spec F3.8 §2
  (`docs/superpowers/specs/2026-07-19-phase-3.8-advanced-showcase-project-details-design.md`).

## Cuándo NO usar este patrón

- Grid uniforme (celdas iguales) → CSS grid `auto-fill/minmax`, sin JS.
- Miles de ítems → añade virtualización (fuera del alcance de este repo: 16 ítems).

## El contexto

Detrás de un catálogo de millones de recursos gráficos hay una fábrica:
miles de colaboradores subiendo contenido que alguien tiene que revisar,
catalogar, moderar y publicar — y toda una operación interna de
producción propia. El backoffice de contenido de Freepik/Flaticon es ese
conjunto de sistemas internos, y este proyecto cubre el trabajo
full-stack en las piezas que lo componen.

## Revisión, catalogación y publicación

El corazón del backoffice: las herramientas con las que el equipo de
contenido revisa lo que llega, lo cataloga (metadatos, etiquetas,
categorías), modera lo que no cumple las normas y publica lo aprobado.
Conviven dos generaciones — un admin histórico y su sucesor sobre
Laravel + Nova — y parte del trabajo fue precisamente esa transición:
mover flujos al sistema nuevo sin parar la fábrica.

## Workflow de producción de contenido

La producción propia (ilustración, vector, PSD, 3D y, más recientemente,
contenido generado con IA) se organiza como un workflow de tareas:
encargos que pasan por estados — asignación, producción, revisión,
entrega — con las reglas de cada tipo de contenido codificadas en el
sistema. El objetivo es que la capacidad de producción sea planificable
y medible, no una cola opaca.

## Packs de iconos de Flaticon

Los iconos de Flaticon no viven sueltos: se agrupan en packs coherentes
(mismo estilo, misma familia). La gestión de packs — composición,
consistencia, versionado y publicación — tiene su propia herramienta
dentro del backoffice.

## Portal fiscal de colaboradores

Los colaboradores cobran por su contenido, y eso exige papeleo fiscal en
regla: un portal donde gestionan sus datos y documentos fiscales, con
validaciones que evitan errores antes de que lleguen a facturación.

## Extensiones Chrome internas

Herramientas de productividad para los equipos internos empaquetadas
como extensiones de Chrome: atajos sobre las pantallas del backoffice
que ahorran pasos repetitivos en los flujos de revisión de cada día.
